52: // 旧36街版は、先頭から続けて修了した街の数を、30フェーズの同じ割合に換算する（切り捨て）
53: function readLegacyPhaseCount(): { count: number } | { error: string } {
54:   const read = readStorage(legacyProgressStorageKey);
55:   if (!read.available) return { error: legacyReadError };
56:   if (read.value === null) return { count: 0 };
57:   try {
58:     const parsed: unknown = JSON.parse(read.value);
59:     if (!parsed || typeof parsed !== 'object' || !hasOwn(parsed, 'completedLessonIds')) return { error: legacyReadError };
60:     const lessons = (parsed as { completedLessonIds: unknown }).completedLessonIds;
61:     if (!Array.isArray(lessons) || lessons.some((value) => typeof value !== 'string')) return { error: legacyReadError };
62:     const completedLessons = new Set(lessons);
63:     const isStageCompleted = (stageId: string) => legacyLessonIdsOf(stageId).every((id) => completedLessons.has(id));
64:     const firstUnfinished = stages.findIndex((stage) => !isStageCompleted(stage.id));
65:     const completedStageCount = firstUnfinished < 0 ? stages.length : firstUnfinished;
66:     return { count: Math.floor(completedStageCount * phases.length / stages.length) };
67:   } catch {
68:     return { error: legacyReadError };
69:   }
70: }

72: // 形が少しでも違う記録は直さずに捨てる（null）。呼び出し側は上書きせずに止める
73: function parseWorldProgress(value: unknown): WorldProgress | null {
74:   if (!value || typeof value !== 'object') return null;
75:   const source = value as Partial<WorldProgress>;
76:   if (source.schemaVersion !== schemaVersion || !Array.isArray(source.completedPhaseIds) || !isPhasePrefix(source.completedPhaseIds)) return null;
77:   const migration = source.migration;
78:   if (!migration || typeof migration !== 'object' || typeof migration.noticeDismissed !== 'boolean') return null;
79:   const legacyCount = migration.legacyPhaseCount;
80:   if (!Number.isInteger(legacyCount) || legacyCount < 0 || legacyCount > source.completedPhaseIds.length) return null;
81:   return {
82:     schemaVersion,
83:     completedPhaseIds: source.completedPhaseIds,
84:     migration: { legacyPhaseCount: legacyCount, noticeDismissed: migration.noticeDismissed },
85:   };
86: }

107: export function loadOrCreateWorldProgress(): LoadResult {
108:   const read = readStorage(worldProgressKey);
109:   if (!read.available) return { ok: false, error: unavailableError };
110:   if (read.value !== null) {
111:     const progress = parseStored(read.value);
112:     return progress ? { ok: true, progress } : { ok: false, error: unreadableMessage };
113:   }
114:

116:   if ('error' in legacy) return { ok: false, error: legacy.error };
117:   const progress: WorldProgress = {
118:     schemaVersion,
119:     completedPhaseIds: phases.slice(0, legacy.count).map((phase) => phase.id),
120:     migration: { legacyPhaseCount: legacy.count, noticeDismissed: false },
121:   };
122:   if (!saveWorldProgress(progress)) return { ok: false, error: saveErrorMessage };
123:   return { ok: true, progress };
